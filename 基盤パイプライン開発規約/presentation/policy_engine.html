# SEC-04: Public access check
deny[msg] {
  r := resource_changes[_]
  public_member(r)
  not is_exception(r)
  msg := sprintf("%s: public principal is prohibited", [r.address])
}

# COST-04: Label check
deny[msg] {
  r := resource_changes[_]
  missing := required_labels - {k | r.change.after.labels[k] != null}
  count(missing) > 0
  msg := sprintf("missing labels %v", [missing])
}
